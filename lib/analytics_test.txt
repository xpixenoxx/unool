CREATE TABLE IF NOT EXISTS platform_insights (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  platform TEXT NOT NULL CHECK (platform IN ('instagram', 'linkedin')),
  platform_post_id UUID NOT NULL,
  metric_date DATE NOT NULL,
  
  -- Reach metrics (platform-native definitions)
  impressions INTEGER NOT NULL DEFAULT 0,        -- Total impressions
  unique_impressions INTEGER NOT NULL DEFAULT 0, -- Unique reach
  
  -- Engagement metrics
  likes INTEGER NOT NULL DEFAULT 0,
  comments INTEGER NOT NULL DEFAULT 0,
  shares INTEGER NOT NULL DEFAULT 0,
  saves INTEGER NOT NULL DEFAULT 0,             -- Instagram only
  
  -- Platform-specific video metrics
  video_views INTEGER,                          -- LinkedIn (min 3s)
  reel_plays INTEGER,                           -- Instagram
  avg_watch_time INTEGER,                       -- Instagram Reels (seconds)
  completion_rate DECIMAL(5,2),                 -- Instagram Reels (%)
  
  -- Audience demographics (JSONB for flexibility)
  age_range_distribution JSONB NOT NULL DEFAULT '{}',   -- {"18-24": 45, "25-34": 32, ...}
  gender_distribution JSONB NOT NULL DEFAULT '{}',      -- {"male": 120, "female": 115}
  top_locations JSONB NOT NULL DEFAULT '{}',          -- {"New York": 23, "LA": 12}
  
  -- Platform-native raw data
  native_data JSONB NOT NULL DEFAULT '{}',
  
  -- Timestamps
  captured_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE(workspace_id, platform, platform_post_id, metric_date)
);

/**
 * Platform Analytics Service - Instagram & LinkedIn
 * Handles webhook reception, metric calculation, and data storage
 */

type Platform = 'instagram' | 'linkedin';

type AgeRange = '13-17' | '18-24' | '25-34' | '35-44' | '45-54' | '55+';
type Gender = 'male' | 'female' | 'non-binary' | 'unknown';

interface InstagramAudience {
  age_ranges: Record<AgeRange, number>;
  gender_distribution: Record<Gender, number>;
  top_locations: Record<string, number>; // city -> count
  active_times: Record<string, number>; // hour -> session count
}

interface LinkedInAudience {
  company_sizes: Record<string, number>; // "1-10", "11-50", etc.
  job_functions: Record<string, number>; // "CEO", "CTO", etc.
  seniorities: Record<string, number>; // "entry", "mid", "senior", "exec"
  geographies: Record<string, number>; // country -> count
  industries: Record<string, number>; // industry codes -> count
}

interface BaseMetrics {
  impressions: number;
  unique_impressions: number;
  likes: number;
  comments: number;
  shares: number;
}

interface InstagramMetrics extends BaseMetrics {
  saves: number;
  reel_plays: number;
  avg_watch_time: number;
  completion_rate: number;
  audience: InstagramAudience;
  hashtag_performance: Record<string, { reach: number; impressions: number }>;
}

interface LinkedInMetrics extends BaseMetrics {
  video_views: number;
  avg_watch_time: number;
  demographic: LinkedInAudience;
  content_type_performance: Record<'article' | 'post' | 'comment', { impressions: number; engagements: number; followers: number }>;
}

interface PlatformMetrics<
  T extends BaseMetrics = BaseMetrics
> extends BaseMetrics {
  platform: Platform;
  metrics: T;
  captured_at: Date;
}

class PlatformAnalyticsService {
  
  /** 
   * Parse Instagram webhook payload into normalized metrics
   * Instagram sends data via webhook when subscribed with post_analytics, story_analytics, etc.
   */
  parseInstagramWebhook(payload: any): InstagramMetrics {
    const { entry } = payload;
    if (!entry || entry.length === 0) {
      throw new Error('Invalid Instagram webhook payload');
    }
    
    const changes = entry[0].changes;
    const results: InstagramMetrics[] = [];
    
    for (const change of changes) {
      const { value, field } = change;
      
      if (field === 'post_analytics') {
        results.push(this.parsePostAnalytics(value));
      } else if (field === 'story_analytics') {
        results.push(this.parseStoryAnalytics(value));
      } else if (field === 'follower_growth') {
        // Handle follower changes separately
      }
    }
    
    return results;
  }
  
  /** Parse post analytics from Instagram Graph API response */
  private parsePostAnalytics(igData: any): InstagramMetrics {
    return {
      platform: 'instagram' as const,
      platform_post_id: igData.id,
      metric_date: new Date().toISOString().slice(0, 10),
      
      // Reach
      impressions: igData.metrics?.impressions?.count ?? 0,
      unique_impressions: igData.metrics?.reach?.count ?? 0,
      
      // Engagement
      likes: igData.metrics?.likes?.count ?? 0,
      comments: igData.metrics?.comments?.count ?? 0,
      shares: igData.metrics?.shares?.count ?? 0,
      saves: igData.metrics?.saves?.count ?? 0,
      
      // Video/Reels
      reel_plays: igData.metrics?.video_plays?.count ?? 0,
      avg_watch_time: igData.metrics?.avg_watch_time ?? 0,
      completion_rate: igData.metrics?.completion_rate ?? 0,
      
      // Audience
      audience: this.parseAudienceData(igData.audience),
      
      // Hashtags
      hashtag_performance: this.parseHashtagPerformance(igData.hashtag_insights),
      
      // Timestamps
      captured_at: new Date()
    };
  }
  
  /** Parse story analytics */
  private parseStoryAnalytics(igData: any): InstagramMetrics {
    return {
      platform: 'instagram' as const,
      platform_post_id: igData.id,
      metric_date: new Date().toISOString().slice(0, 10),
      
      impressions: igData.metrics?.impressions?.count ?? 0,
      unique_impressions: igData.metrics?.reach?.count ?? 0,
      
      // Story-specific
      // (story metrics structure differs - omitted for brevity)
      
      audience: this.parseAudienceData(igData.audience),
      hashtag_performance: {},
      captured_at: new Date()
    };
  }
  
  /** Parse audience data into structured format */
  private parseAudienceData(audience: any): InstagramAudience {
    return {
      age_ranges: this.mapAgeRanges(audience?.age_distribution),
      gender_distribution: this.mapGenderDistribution(audience?.gender_distribution),
      top_locations: audience?.top_locations?.reduce(
        (acc: Record<string, number>, loc: any) => {
          acc[loc.name] = (acc[loc.name] || 0) + loc.count;
          return acc;
        }, {}
      ),
      active_times: this.parseActiveTimes(audience?.active_times)
    };
  }
  
  /** Parse hashtag performance */
  private parseHashtagPerformance(hashtags: any): Record<string, { reach: number; impressions: number }> {
    const result: Record<string, { reach: number; impressions: number }> = {};
    
    if (hashtags && hashtags.data) {
      for (const ht of hashtags.data) {
        result[ht.name] = {
          reach: ht.reach?.count ?? 0,
          impressions: ht.impressions?.count ?? 0
        };
      }
    }
    
    return result;
  }
  
  /** Map raw age distribution to our enum keys */
  private mapAgeRanges(raw: any): Record<AgeRange, number> {
    const ranges: Record<string, AgeRange> = {
      '13-17': '13-17',
      '18-24': '18-24', 
      '25-34': '25-34',
      '35-44': '35-44',
      '45-54': '45-54',
      '55+': '55+'
    };
    
    const result: Record<AgeRange, number> = {
      '13-17': 0, '18-24': 0, '25-34': 0, '35-44': 0, '45-54': 0, '55+': 0
    };
    
    if (raw) {
      for (const [key, count] of Object.entries(raw)) {
        if (ranges[key] && typeof count === 'number') {
          result[ranges[key]] = count;
        }
      }
    }
    
    return result;
  }
  
  /** Map gender distribution */
  private mapGenderDistribution(raw: any): Record<Gender, number> {
    const result: Record<Gender, number> = { male: 0, female: 0, non_binary: 0, unknown: 0 };
    
    if (raw) {
      if (raw.male && typeof raw.male === 'number') result.male = raw.male;
      if (raw.female && typeof raw.female === 'number') result.female = raw.female;
      if (raw.non_binary && typeof raw.non_binary === 'number') result.non_binary = raw.non_binary;
      if (raw.unknown && typeof raw.unknown === 'number') result.unknown = raw.unknown;
    }
    
    return result;
  }
  
  /** Parse active times into hour-based format */
  private parseActiveTimes(raw: any): Record<string, number> {
    const result: Record<string, number> = {};
    
    if (raw && raw.hours) {
      for (const [hour, data] of Object.entries(raw.hours)) {
        // Instagram reports hour in user's timezone
        const h = parseInt(hour, 10);
        result[`${h}:00`] = data?.count ?? 0;
      }
    }
    
    return result;
  }
}

class LinkedInAnalyticsParser {
  
  parseLinkedInWebhook(payload: any): LinkedInMetrics {
    const { ritualId, metrics, author } = payload;
    
    return {
      platform: 'linkedin' as const,
      platform_post_id: ritualId,
      metric_date: new Date().toISOString().slice(0, 10),
      
      impressions: metrics?.servedImpressions ?? 0,
      unique_impressions: metrics?.uniqueImpressions ?? 0,
      likes: metrics?.reactions?.totalCount ?? 0,
      comments: metrics?.comments?.count ?? 0,
      shares: metrics?.socialAnnotations?.shareCount ?? 0,
      video_views: metrics?.video?.viewCount ?? 0,
      avg_watch_time: metrics?.video?.averageWatchTimeMs 
        ? Math.floor(metrics.video.averageWatchTimeMs / 1000) 
        : 0,
      
      // Audience demographics
      demographic: this.parseLinkedInAudience(author?.audience),
      
      // Content type performance
      content_type_performance: this.parseContentTypePerformance(author?.content),
      
      captured_at: new Date()
    };
  }
  
  /** Parse LinkedIn audience demographics */
  private parseLinkedInAudience(author: any): LinkedInAudience {
    if (!author) {
      return {
        company_sizes: {},
        job_functions: {},
        seniorities: {},
        geographies: {},
        industries: {}
      };
    }
    
    return {
      company_sizes: this.mapCompanySizes(author.audience?.company_sizes),
      job_functions: this.mapJobFunctions(author.audience?.job_functions),
      seniorities: this.mapSeniorities(author.audience?.seniorities),
      geographies: this.mapGeographies(author.audience?.geographic_distribution),
      industries: this.mapIndustries(author.audience?.industry_distribution)
    };
  }
  
  /** Parse content type performance from author stats */
  private parseContentTypePerformance(author: any): Record<'article' | 'post' | 'comment', { impressions: number; engagements: number; followers: number }> {
    const result = {
      article: { impressions: 0, engagements: 0, followers: 0 },
      post: { impressions: 0, engagements: 0, followers: 0 },
      comment: { impressions: 0, engagements: 0, followers: 0 }
    };
    
    if (author?.content_performance) {
      for (const [type, data] of Object.entries(author.content_performance)) {
        const normalizedType = type as 'article' | 'post' | 'comment';
        result[normalizedType] = {
          impressions: data.impressions?.total ?? 0,
          engagements: data.total Engagement ?? 0,
          followers: data.followerGain ?? 0
        };
      }
    }
    
    return result;
  }
  
  /** Map company sizes from LinkedIn's format */
  private mapCompanySizes(raw: any): Record<string, number> {
    const mapping: Record<string, string> = {
      '1-10': '1-10',
      '11-50': '11-50', 
      '51-200': '51-200',
      '201-500': '201-500',
      '501-1k': '501-1k',
      '1k-5k': '1k-5k',
      '5k-10k': '5k-10k',
      '10k+': '10k+'
    };
    
    const result: Record<string, number> = {
      '1-10': 0, '11-50': 0, '51-200': 0, '201-500': 0,
      '501-1k': 0, '1k-5k': 0, '5k-10k': 0, '10k+': 0
    };
    
    if (raw) {
      for (const [size, count] of Object.entries(raw)) {
        const normalized = mapping[size] || size;
        if (typeof count === 'number') {
          result[normalized] = count;
        }
      }
    }
    
    return result;
  }
  
  /** Map job functions */
  private mapJobFunctions(raw: any): Record<string, number> {
    const result: Record<string, number> = {};
    if (raw) {
      for (const [func, count] of Object.entries(raw)) {
        if (typeof count === 'number') {
          result[func] = count;
        }
      }
    }
    return result;
  }
  
  /** Map seniorities */
  private mapSeniorities(raw: any): Record<string, number> {
    const mapping: Record<string, string> = {
      'entry-level': 'entry',
      'associate': 'entry',
      'mid-career': 'mid',
      'manager': 'mid',
      'director': 'senior',
      'vp': 'senior',
      'c-suite': 'exec',
      'head': 'exec'
    };
    
    const result: Record<string, number> = {
      entry: 0, mid: 0, senior: 0, exec: 0
    };
    
    if (raw) {
      for (const [seniority, count] of Object.entries(raw)) {
        const normalized = mapping[seniority] || seniority;
        if (typeof count === 'number') {
          result[normalized] = count;
        }
      }
    }
    
    return result;
  }
  
  /** Map geographic distribution */
  private mapGeographies(raw: any): Record<string, number> {
    const result: Record<string, number> = {};
    if (raw) {
      for (const [country, count] of Object.entries(raw)) {
        if (typeof count === 'number') {
          result[country] = count;
        }
      }
    }
    return result;
  }
  
  /** Map industry distribution */
  private mapIndustries(raw: any): Record<string, number> {

      for (const [industry, count] of Object.entries(raw)) {

/**
 * Calculate derived metrics from raw platform data
 */
class MetricsCalculator {
  
  /** Calculate engagement rate per 1,000 impressions */
  static calculateEngagementRate(impressions: number, likes: number, comments: number, shares: number): number {
    if (impressions === 0) return 0;
    const totalEngagement = likes + comments + shares;
    return Math.round((totalEngagement / impressions) * 1000);
  }
  
  /** Calculate virality rate (shares/impressions * 100) */
  static calculateViralityRate(impressions: number, shares: number): number {
    if (impressions === 0) return 0;
    return Math.round((shares / impressions) * 100);
  }
  
  /** Calculate click-through rate (link clicks/impressions * 100) */
  static calculateClickThroughRate(impressions: number, linkClicks: number): number {
    if (impressions === 0) return 0;
    return Math.round((linkClicks / impressions) * 100);
  }
  
  /** Calculate save rate (saves/reach * 100) for Instagram */
  static calculateSaveRate(reach: number, saves: number): number {
    if (reach === 0) return 0;
    return Math.round((saves / reach) * 100);
  }
  
  /** Calculate growth rate (new followers - unfollowers) / total followers */
  static calculateGrowthRate(newFollowers: number, unfollowers: number, totalFollowers: number): number {
    if (totalFollowers === 0) return 0;
    const net = newFollowers - unfollowers;
    return Math.round((net / totalFollowers) * 100);
  }
  
  /** Calculate resonance score (0-100) based on multiple factors */
  static calculateResonanceScore({
    engagementRate,     // per 1000
    viralityRate,       // %
    saveRate,           // %
    commentRate,        // comments/impressions * 100
    postCount: number   // number of posts in period
  }: {
    engagementRate: number;
    viralityRate: number;
    saveRate: number;
    commentRate: number;
    postCount: number;
  }): number {
    // Weighted formula - each factor contributes to 0-100 score
    const weights = {
      engagementRate: 0.25,    // 25 points
      viralityRate: 0.20,      // 20 points
      saveRate: 0.20,          // 20 points
      commentRate: 0.20,       // 20 points
      postCount: 0.15          // 15 points (diminishing returns)
    };
    
    // Normalize each component to 0-100 scale
    const normEngagement = Math.min(engagementRate / 10, 100); // ~2.5%+ is excellent
    const normVirality = Math.min(viralityRate, 100);
    const normSave = Math.min(saveRate, 100);
    const normComment = Math.min(commentRate * 50, 100); // ~2% comment rate normalized
    
    // Post count factor (more posts = more data, but diminishing)
    const postFactor = Math.min(postCount / 20, 1) * 15;
    
    const score = (
      normEngagement * weights.engagementRate +
      normVirality * weights.viralityRate +
      normSave * weights.saveRate +
      normComment * weights.commentRate +
      postFactor
    );
    
    return Math.round(Math.min(score, 100));
  }
  
  /** Determine hook type performance from post variants */
  static analyzeHookPerformance(variants: any[]): Record<string, { count: number; avgEngagement: number; topPerformers: number[] }> {
    const hookTypes = ['statement', 'question', 'statistic', 'story', 'contrarian', 'framework'];
    const result: Record<string, { count: number; avgEngagement: number; topPerformers: number[] }> = {};
    
    for (const type of hookTypes) {
      result[type] = { count: 0, avgEngagement: 0, topPerformers: [] };
    }
    
    for (const variant of variants) {
      const type = variant.hookType || 'statement';
      if (!result[type]) continue;
      
      result[type].count++;
      const engagement = variant.engagement?.totalInteractions ?? 0;
      result[type].avgEngagement = (
        (result[type].avgEngagement * (result[type].count - 1) + engagement) / result[type].count
      );
      
      // Top performers: top 20% by engagement
      if (engagement > 0 && engagement >= 50) { // threshold adjustable
        result[type].topPerformers.push(variant.id);
      }
    }
    
    return result;
  }
  
  /** Find optimal posting time from audience activity data */
  static findOptimalPostingTime(audienceActiveHours: Record<string, number>): string {
    if (!audienceActiveHours || Object.keys(audienceActiveHours).length === 0) {
      return '10:00'; // default
    }
    
    // Find hour with most activity
    let maxHour = 0;
    let maxCount = 0;
    
    for (const [hourStr, count] of Object.entries(audienceActiveHours)) {
      const hour = parseInt(hourStr, 10);
      if (count > maxCount) {
        maxCount = count;
        maxHour = hour;
      }
    }
    
    // Return in 12-hour format with AM/PM
    if (maxHour === 0) return '10:00 AM';
    
    const hour12 = maxHour % 12 || 12;
    const amPm = maxHour < 12 ? 'AM' : 'PM';
    
    return `${hour12}:00 ${amPm}`;
  }
  
  /** Determine optimal word count range from performance data */
  static findOptimalWordCountRange(variants: any[]): { min: number; max: number; bestRange: [number, number] } {
    // Group posts by word count buckets
    const buckets: Record<string, { count: number; totalEngagement: number }> = {
      'micro': { count: 0, totalEngagement: 0 },    // 0-50 words
      'short': { count: 0, totalEngagement: 0 },    // 51-80 words
      'optimal': { count: 0, totalEngagement: 0 },  // 81-120 words
      'long': { count: 0, totalEngagement: 0 },     // 121-200 words
      'very-long': { count: 0, totalEngagement: 0 } // 200+ words
    };
    
    const wordRanges = [
      { min: 0, max: 50, key: 'micro' },
      { min: 51, max: 80, key: 'short' },
      { min: 81, max: 120, key: 'optimal' },
      { min: 121, max: 200, key: 'long' },
      { min: 201, max: Infinity, key: 'very-long' }
    ];
    
    for (const variant of variants) {
      const wordCount = variant.content?.split(/\s+/).filter(w => w.length > 0).length ?? 0;
      const engagement = variant.engagement?.totalInteractions ?? 0;
      
      for (const bucket of buckets) {
        if (wordCount >= bucket.min && wordCount <= bucket.max) {
          bucket.count++;
          bucket.totalEngagement += engagement;
          break;
        }
      }
    }
    
    // Find best performing bucket
    let bestBucket = 'optimal';
    let bestAvgEngagement = 0;
    
    for (const [key, bucket] of Object.entries(buckets)) {
      const avg = bucket.count > 0 ? bucket.totalEngagement / bucket.count : 0;
      if (avg > bestAvgEngagement) {
        bestAvgEngagement = avg;
        bestBucket = key;
      }
    }
    
    // Return the best bucket's range plus adjacent range
    const bestBucketData = buckets[bestBucket];
    const bestRange: [number, number] = bestBucketData.count > 0 
      ? [bestBucketData.min, bestBucketData.max] 
      : [81, 120]; // default
    
    return { min: bestRange[0], max: bestRange[1], bestRange };
  }
}

// app/api/webhooks/instagram/route.ts
import { NextResponse } from 'next/server';
import { verifyInstagramWebhook } from '@/lib/security';
import { PlatformAnalyticsService } from '@/lib/analytics/platform-analytics';

export async function POST(request: Request) {
  try {
    const payload = await request.json();
    
    // Verify webhook signature for security
    const isValid = await verifyInstagramWebhook(request);
    if (!isValid) {
      return NextResponse.json({ error: 'Invalid signature' }, { status: 401 });
    }
    
    const analyticsService = new PlatformAnalyticsService();
    const metrics = analyticsService.parseInstagramWebhook(payload);
    
    // Store each metric set in database
    const results = await Promise.all(metrics.map(metric =>
      storePlatformInsight(metric)
    ));
    
    return NextResponse.json({
      success: true,
      processed: metrics.length,
      results
    });
  } catch (error) {
    console.error('Instagram webhook error:', error);
    return NextResponse.json({ error: 'Processing failed' }, { status: 500 });
  }
}

      table: 'platform_insights',

    }, (payload) => {

    })

5. AI INSIGHTS ENGINE (Deterministic, No Hallucinations)
// lib/insights/engine.ts - DETERMINISTIC rules only

export type InsightType = 
  | 'hook_performance' 
  | 'optimal_timing' 
  | 'content_format' 
  | 'hashtag_recommendation'
  | 'growth_pattern';

export type InsightSeverity = 'info' | 'warning' | 'recommendation';

export interface Insight {
  id: string;
  type: InsightType;
  severity: InsightSeverity;
  title: string;
  description: string;
  action: string;  // "Do this" or "Stop doing this"
  metricImpact: {
    current: number;
    potential: number;
    unit: string;
  };
  confidence: 0.9 | 0.8 | 0.7; // Based on data volume
  generatedAt: Date;
}

export class InsightsEngine {
  
  /** Analyze hook type performance across user's posts */
  static async analyzeHookPerformance(
    workspaceId: string,
    postCount: number = 30
  ): Promise<Insight[]> {
    const insights: Insight[] = [];
    
    // Fetch user's post variants from last N posts
    const variants = await fetchUserVariants(workspaceId, postCount);
    
    if (variants.length === 0) {
      return [{ id: crypto.randomUUID(), type: 'hook_performance', severity: 'info', 
        title: 'Build Your Hook Library', 
        description: 'Start posting to gather data on hook performance',
        action: 'Post your first content with different hook types',
        metricImpact: { current: 0, potential: 0, unit: '' },
        confidence: 0.3,
        generatedAt: new Date()
      }];
    }
    
    // Analyze each hook type
    const performance = MetricsCalculator.analyzeHookPerformance(variants);
    
    // Check if Question hooks are performing significantly better
    const questionPerf = performance.question;
    const statementPerf = performance.statement;
    
    if (questionPerf.count > 5 && statementPerf.count > 5) {
      const questionAvg = questionPerf.avgEngagement;
      const statementAvg = statementPerf.avgEngagement;
      
      if (questionAvg > statementAvg * 1.5) {
        insights.push({
          id: crypto.randomUUID(),
          type: 'hook_performance',
          severity: 'recommendation',
          title: 'Question Hooks Are Your Winner',
          description: `Your Question hooks average ${questionAvg.toFixed(0)} engagement vs ${statementAvg.toFixed(0)} for Statement hooks - a ${((questionAvg/statementAvg - 1) * 100).toFixed(0)}% difference`,
          action: 'Prioritize Question hooks in your next 5 posts',
          metricImpact: { 
            current: statementAvg, 
            potential: questionAvg, 
            unit: 'engagement per post' 
          },
          confidence: 0.95
        });
      }
    }
    
    // Similar analysis for other hook types...
    
    // Optimal timing insight
    const timingInsight = await this.analyzeOptimalTiming(variants);
    insights.push(timingInsight);
    
    // Content format insight
    const formatInsight = this.analyzeContentFormat(variants);
    insights.push(formatInsight);
    
    // Sort by severity (recommendations first, then info)
    return insights.sort((a, b) => {
      const severityOrder: Record<string, number> = { recommendation: 1, info: 0, warning: 2 };
      return (severityOrder[b.severity] || 0) - (severityOrder[a.severity] || 0);
    });
  }
  
  /** Analyze optimal posting time */
  private static async analyzeOptimalTiming(variants: any[]): Promise<Insight> {
    // Group by hour of day
    const hourData: Record<string, { count: number; totalEngagement: number }> = {};
    
    for (const variant of variants) {
      const publishHour = new Date(variant.createdAt).getHours();
      const hourKey = `${publishHour}:00`;
      
      if (!hourData[hourKey]) {
        hourData[hourKey] = { count: 0, totalEngagement: 0 };
      }
      
      hourData[hourKey].count++;
      hourData[hourKey].totalEngagement += variant.engagement?.totalInteractions ?? 0;
    }
    
    // Find best hour
    let bestHour = '10:00';
    let bestAvgEngagement = 0;
    
    for (const [hour, data] of Object.entries(hourData)) {
      const avg = data.count > 0 ? data.totalEngagement / data.count : 0;
      if (avg > bestAvgEngagement) {
        bestAvgEngagement = avg;
        bestHour = hour;
      }
    }
    
    return {
      id: crypto.randomUUID(),
      type: 'optimal_timing',
      severity: 'recommendation',
      title: 'Your Optimal Posting Time',
      description: `Your audience is most active at ${bestHour} based on your post performance data`,
      action: `Post your next content at ${bestHour} for maximum visibility`,
      metricImpact: { 
        current: 0, 
        potential: bestAvgEngagement, 
        unit: 'engagement per post' 
      },
      confidence: hourData[bestHour] ? 0.85 : 0.3,
      generatedAt: new Date()
    };
  }
  
  /** Analyze content format performance */
  private static analyzeContentFormat(variants: any[]): Insight {
    const formatData: Record<'carousel' | 'single' | 'reel', { count: number; totalEngagement: number }> = {
      carousel: { count: 0, totalEngagement: 0 },
      single: { count: 0, totalEngagement: 0 },
      reel: { count: 0, totalEngagement: 0 }
    };
    
    for (const variant of variants) {
      const mediaCount = variant.mediaUrls?.length ?? 1;
      const type = mediaCount >= 3 ? 'carousel' : mediaCount === 1 && variant.mediaUrls?.[0]?.includes('video') || variant.media?.format === 'video' ? 'reel' : 'single';
      
      if (!formatData[type]) continue;
      
      formatData[type].count++;
      formatData[type].totalEngagement += variant.engagement?.totalInteractions ?? 0;
    }
    
    // Find best format
    let bestFormat = 'single';
    let bestAvg = 0;
    
    for (const [format, data] of Object.entries(formatData)) {
      const avg = data.count > 0 ? data.totalEngagement / data.count : 0;
      if (avg > bestAvg) {
        bestAvg = avg;
        bestFormat = format;
      }
    }
    
    const formatNames: Record<'carousel' | 'single' | 'reel', string> = {
      carousel: 'Carousel Posts',
      single: 'Single Image/Video',
      reel: 'Reels'
    };
    
    let action = 'Continue using your current format';
    let description = `Your ${formatNames[bestFormat]} posts perform best`;
    
    if (bestAvg > 0) {
      // Compare to worst format
      const worstFormat = Object.keys(formatData).find(k => k !== bestFormat);
      const worstAvg = formatData[worstFormat]?.count > 0 
        ? formatData[worstFormat].totalEngagement / formatData[worstFormat].count 
        : 0;
      
      if (bestAvg > worstAvg * 1.5) {
        action = `Shift toward ${formatNames[bestFormat]} - they get ${((bestAvg / worstAvg - 1) * 100).toFixed(0)}% better engagement`;
        description = `Your ${formatNames[bestFormat]} posts average ${bestAvg.toFixed(0)} engagement vs ${worstAvg.toFixed(0)} for ${formatNames[worstFormat]}`;
      }
    }
    
    return {
      id: crypto.randomUUID(),
      type: 'content_format',
      severity: 'recommendation',
      title: `${formatNames[bestFormat]} Are Your Winner`,
      description,
      action,
      metricImpact: { 
        current: 0, 
        potential: bestAvg, 
        unit: 'engagement per post' 
      },
      confidence: formatData[bestFormat]?.count > 5 ? 0.9 : 0.5,
      generatedAt: new Date()
    };
  }
  
  /** Get all insights for a workspace */
  static async getWorkspaceInsights(workspaceId: string): Promise<Insight[]> {
    // Fetch data from last 30 days
    const variants = await fetchUserVariants(workspaceId, 90);
    const posts = await fetchUserPosts(workspaceId, 90);
    
    // Run all insight analyses
    const hookInsights = await this.analyzeHookPerformance(workspaceId, 30);
    const timingInsight = await this.analyzeOptimalTiming(variants);
    const formatInsight = this.analyzeContentFormat(variants);
    
    // Also calculate resonance score
    const resonance = MetricsCalculator.calculateResonanceScore({
      engagementRate: 0, // would calculate from data
      viralityRate: 0,
      saveRate: 0,
      commentRate: 0,
      postCount: variants.length
    });
    
    if (resonance > 80) {
      insights.push({
        id: crypto.randomUUID(),
        type: 'growth_pattern',
        severity: 'info',
        title: 'Strong Resonance Detected',
        description: `Your resonance score is ${resonance} - your content strategy is working well. Keep applying what you're doing!`,
        action: 'Continue your current content strategy',
        metricImpact: { current: 0, potential: 0, unit: '' },
        confidence: 0.9,
        generatedAt: new Date()
      });
    }
    
    return [...hookInsights, timingInsight, formatInsight].sort((a, b) => {
      const severityOrder: Record<string, number> = { recommendation: 1, info: 0, warning: 2 };
      return (severityOrder[b.severity] || 0) - (severityOrder[a.severity] || 0);
    });
  }
}

  const isDown = changeWoW !== undefined && changeWoW < 0;

      className="rounded-2xl p-6 bg-white border border-border hover:shadow-lg transition-shadow duration-200"

class PlatformAnalyticsService {

  parseInstagramWebhook(payload: any): InstagramMetrics[] {
    const { entry } = payload;
    if (!entry || entry.length === 0) throw new Error('Invalid Instagram webhook payload');
    
    const changes = entry[0].changes;
    const results: InstagramMetrics[] = [];
    
    for (const change of changes) {
      const { value, field } = change;
      if (field === 'post_analytics') {
        results.push(this.parsePostAnalytics(value));
      } else if (field === 'story_analytics') {
        results.push(this.parseStoryAnalytics(value));
      }
    }
    return results;
  }

  private parsePostAnalytics(igData: any): InstagramMetrics {
    return {
      platform: 'instagram' as const,
      platform_post_id: igData.id,
      metric_date: new Date().toISOString().slice(0, 10),
      impressions: igData.metrics?.impressions?.count ?? 0,
      unique_impressions: igData.metrics?.reach?.count ?? 0,
      likes: igData.metrics?.likes?.count ?? 0,
      comments: igData.metrics?.comments?.count ?? 0,
      shares: igData.metrics?.shares?.count ?? 0,
      saves: igData.metrics?.saves?.count ?? 0,
      reel_plays: igData.metrics?.video_plays?.count ?? 0,
      avg_watch_time: igData.metrics?.avg_watch_time ?? 0,
      completion_rate: igData.metrics?.completion_rate ?? 0,
      audience: this.parseAudienceData(igData.audience),
      hashtag_performance: this.parseHashtagPerformance(igData.hashtag_insights),
      captured_at: new Date()
    };
  }

  private parseStoryAnalytics(igData: any): InstagramMetrics {
    return {
      platform: 'instagram' as const,
      platform_post_id: igData.id,
      metric_date: new Date().toISOString().slice(0, 10),
      impressions: igData.metrics?.impressions?.count ?? 0,
      unique_impressions: igData.metrics?.reach?.count ?? 0,
      audience: this.parseAudienceData(igData.audience),
      hashtag_performance: {},
      captured_at: new Date()
    };
  }

  private parseAudienceData(audience: any): InstagramAudience {
    return {
      age_ranges: this.mapAgeRanges(audience?.age_distribution),
      gender_distribution: this.mapGenderDistribution(audience?.gender_distribution),
      top_locations: audience?.top_locations?.reduce((acc: Record<string, number>, loc: any) => {
        acc[loc.name] = (acc[loc.name] || 0) + loc.count;
        return acc;
      }, {}),
      active_times: this.parseActiveTimes(audience?.active_times)
    };
  }

  private parseHashtagPerformance(hashtags: any): Record<string, { reach: number; impressions: number }> {
    const result: Record<string, { reach: number; impressions: number }> = {};
    if (hashtags && hashtags.data) {
      for (const ht of hashtags.data) {
        result[ht.name] = { reach: ht.reach?.count ?? 0, impressions: ht.impressions?.count ?? 0 };
      }
    }
    return result;
  }

  private mapAgeRanges(raw: any): Record<AgeRange, number> {
    const ranges: Record<string, AgeRange> = {
      '13-17': '13-17', '18-24': '18-24', '25-34': '25-34',
      '35-44': '35-44', '45-54': '45-54', '55+': '55+'
    };
    const result: Record<AgeRange, number> = { '13-17': 0, '18-24': 0, '25-34': 0, '35-44': 0, '45-54': 0, '55+': 0 };
    if (raw) {
      for (const [key, count] of Object.entries(raw)) {
        if (ranges[key] && typeof count === 'number') result[ranges[key]] = count;
      }
    }
    return result;
  }

    const result: Record<Gender, number> = { male: 0, female: 0, non_binary: 0, unknown: 0 };
    if (raw) {

      if (raw.female && typeof raw.female === 'number') result.female = raw.female;

  if (!secret || !signature) return false;
  const [algo, receivedHash] = signature.split('=');

  const crypto = require('crypto');

  return localHash === receivedHash;
}

export type InsightType = 'hook_performance' | 'optimal_timing' | 'content_format' | 'hashtag_recommendation' | 'growth_pattern';
export type InsightSeverity = 'info' | 'warning' | 'recommendation';

export interface Insight {
  id: string;
  type: InsightType;
  severity: InsightSeverity;
  title: string;
  description: string;
  action: string;
  metricImpact: { current: number; potential: number; unit: string };
  confidence: 0.9 | 0.8 | 0.7;
  generatedAt: Date;
}

export class InsightsEngine {
  static async analyzeHookPerformance(workspaceId: string, postCount: number = 30): Promise<Insight[]> {
    const insights: Insight[] = [];
    const variants = await fetchUserVariants(workspaceId, postCount);
    
    if (variants.length === 0) {
      return [{ id: crypto.randomUUID(), type: 'hook_performance', severity: 'info', title: 'Build Your Hook Library', description: 'Start posting to gather data', action: 'Post your first content', metricImpact: { current: 0, potential: 0, unit: '' }, confidence: 0.3, generatedAt: new Date() }];
    }
    
    const performance = MetricsCalculator.analyzeHookPerformance(variants);
    const questionPerf = performance.question;
    const statementPerf = performance.statement;
    
    if (questionPerf.count > 5 && statementPerf.count > 5) {
      const questionAvg = questionPerf.avgEngagement;
      const statementAvg = statementPerf.avgEngagement;
      
      if (questionAvg > statementAvg * 1.5) {
        insights.push({
          id: crypto.randomUUID(), type: 'hook_performance', severity: 'recommendation',
          title: 'Question Hooks Are Your Winner',
          description: `Your Question hooks average ${questionAvg.toFixed(0)} engagement vs ${statementAvg.toFixed(0)} for Statement hooks - a ${((questionAvg/statementAvg - 1) * 100).toFixed(0)}% difference`,
          action: 'Prioritize Question hooks in your next 5 posts',
          metricImpact: { current: statementAvg, potential: questionAvg, unit: 'engagement per post' }, confidence: 0.95, generatedAt: new Date()
        });
      }
    }
    
    const timingInsight = await this.analyzeOptimalTiming(variants);
    insights.push(timingInsight);
    const formatInsight = this.analyzeContentFormat(variants);
    insights.push(formatInsight);
    
    return insights.sort((a, b) => ({ recommendation: 1, info: 0, warning: 2 }[b.severity] || 0) - { recommendation: 1, info: 0, warning: 2 }[a.severity] || 0);
  }

  private static async analyzeOptimalTiming(variants: any[]): Promise<Insight> {
    const hourData: Record<string, { count: number; totalEngagement: number }> = {};
    for (const variant of variants) {
      const publishHour = new Date(variant.createdAt).getHours();
      const hourKey = `${publishHour}:00`;
      if (!hourData[hourKey]) hourData[hourKey] = { count: 0, totalEngagement: 0 };
      hourData[hourKey].count++;
      hourData[hourKey].totalEngagement += variant.engagement?.totalInteractions ?? 0;
    }
    
    let bestHour = '10:00', bestAvgEngagement = 0;
    for (const [hour, data] of Object.entries(hourData)) {
      const avg = data.count > 0 ? data.totalEngagement / data.count : 0;
      if (avg > bestAvgEngagement) { bestAvgEngagement = avg; bestHour = hour; }
    }
    
    return {
      id: crypto.randomUUID(), type: 'optimal_timing', severity: 'recommendation',
      title: 'Your Optimal Posting Time',
      description: `Your audience is most active at ${bestHour} based on your post performance data`,
      action: `Post your next content at ${bestHour} for maximum visibility`,
      metricImpact: { current: 0, potential: bestAvgEngagement, unit: 'engagement per post' }, confidence: hourData[bestHour] ? 0.85 : 0.3, generatedAt: new Date()
    };
  }

  private static analyzeContentFormat(variants: any[]): Insight {
    const formatData = { carousel: { count: 0, totalEngagement: 0 }, single: { count: 0, totalEngagement: 0 }, reel: { count: 0, totalEngagement: 0 } };
    for (const variant of variants) {
      const mediaCount = variant.mediaUrls?.length ?? 1;
      const type = mediaCount >= 3 ? 'carousel' : mediaCount === 1 && (variant.mediaUrls?.[0]?.includes('video') || variant.media?.format === 'video') ? 'reel' : 'single';
      if (!formatData[type]) continue;
      formatData[type].count++;
      formatData[type].totalEngagement += variant.engagement?.totalInteractions ?? 0;
    }
    
    let bestFormat = 'single', bestAvg = 0;
    for (const [format, data] of Object.entries(formatData)) {
      const avg = data.count > 0 ? data.totalEngagement / data.count : 0;
      if (avg > bestAvg) { bestAvg = avg; bestFormat = format; }
    }
    
    const formatNames = { carousel: 'Carousel Posts', single: 'Single Image/Video', reel: 'Reels' };
    let action = 'Continue using your current format';
    let description = `Your ${formatNames[bestFormat]} posts perform best`;
    
    if (bestAvg > 0) {
      const worstFormat = Object.keys(formatData).find(k => k !== bestFormat);
      const worstAvg = formatData[worstFormat]?.count > 0 ? formatData[worstFormat].totalEngagement / formatData[worstFormat].count : 0;
      if (bestAvg > worstAvg * 1.5) {
        action = `Shift toward ${formatNames[bestFormat]} - they get ${((bestAvg / worstAvg - 1) * 100).toFixed(0)}% better engagement`;
        description = `Your ${formatNames[bestFormat]} posts average ${bestAvg.toFixed(0)} engagement vs ${worstAvg.toFixed(0)} for ${formatNames[worstFormat]}`;
      }
    }
    
    return {
      id: crypto.randomUUID(), type: 'content_format', severity: 'recommendation',
      title: `${formatNames[bestFormat]} Are Your Winner`,
      description,
      action,
      metricImpact: { current: 0, potential: bestAvg, unit: 'engagement per post' }, confidence: formatData[bestFormat]?.count > 5 ? 0.9 : 0.5, generatedAt: new Date()
    };
  }

  static async getWorkspaceInsights(workspaceId: string): Promise<Insight[]> {
    const variants = await fetchUserVariants(workspaceId, 90);
    const posts = await fetchUserPosts(workspaceId, 90);

    return [...hookInsights, timingInsight, formatInsight].sort((a, b) => ({ recommendation: 1, info: 0, warning: 2 }[b.severity] || 0) - { recommendation: 1, info: 0, warning: 2 }[a.severity] || 0);
  }

import { motion } from 'framer-motion';
import { cn } from '@/lib/utils';

type KpiCardProps = {
  label: string;
  value: number;
  suffix?: string;
  changeWoW?: number;
  sparklineData?: number[];
  isLoading?: boolean;
};

const KpiCard = ({ label, value, suffix = '', changeWoW, sparklineData, isLoading = false }: KpiCardProps) => {
  const isUp = changeWoW !== undefined && changeWoW > 0;
  const isDown = changeWoW !== undefined && changeWoW < 0;
  const deltaColor = isUp ? 'hsl(142 70% 45%)' : isDown ? 'hsl(0 80% 50%)' : 'hsl(0 0% 50%)';
  const deltaLabel = changeWoW !== undefined ? `${changeWoW > 0 ? '+' : ''}${changeWoW}% vs last week` : 'No data yet';
  
  const sparkPoints = useMemo(() => {
    if (!sparklineData || sparklineData.length < 2) return null;
    const maxVal = Math.max(...sparklineData, 1);
    return sparklineData.map((v, i) => {
      const x = (i / (sparklineData.length - 1)) * 100;
      const y = 100 - (v / maxVal) * 80;
      return `${x},${y}`;
    }).join(' ');
  }, [sparklineData]);
  
  return (
    <motion.div
      className="rounded-2xl p-6 bg-white border border-border hover:shadow-lg transition-shadow duration-200"
      whileHover={{ y: -4, boxShadow: '0 8px 32px rgba(61,43,31,0.15)' }}
    >
      <div className="flex items-center justify-between mb-3">
        <span className="text-xs uppercase tracking-wider text-muted-foreground">{label}</span>
        <div className="flex items-baseline gap-1">
          {changeWoW !== undefined && (
            <span className="ml-2 text-xs font-semibold text-{isUp ? 'green' : 'red'}">
              {isUp ? '+' : ''}{changeWoW}% {deltaLabel}
            </span>
          )}
        </div>
      </div>
      
      <div className="text-4xl font-extrabold tabular-nums text-foreground">
        {isLoading ? '—' : value.toLocaleString(undefined, { maximumFractionDigits: 1 })} {suffix}
      </div>
      
      {sparklineData && (
        <div className="mt-4 h-16 w-full">
          <svg viewBox="0 0 100 100" preserveAspectRatio="none" className="w-full h-full">
            <defs>
              <linearGradient id="kpiGradient" x1="0%" y1="0%" x2="0%" y2="100%">
                <stop offset="0%" stopColor="hsl(39 97% 52%)" stopOpacity="0.25" />
                <stop offset="100%" stopColor="hsl(39 97% 52%)" stopOpacity="0.02" />
              </linearGradient>
            </defs>
            <polyline points={`0,100 ${sparkPoints} 100,100`} fill="url(#kpiGradient)" />
            <polyline points={sparkPoints} fill="none" stroke="hsl(39 97% 52%)" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
          </svg>
        </div>
      )}
      
      {changeWoW !== undefined && (
        <p className="mt-2 text-xs text-muted-foreground">{deltaLabel}</p>
      )}
    </motion.div>
  );
};
